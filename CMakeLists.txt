cmake_minimum_required(VERSION 3.21)
project(CursorBridge VERSION 0.9.2 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "CursorBridge currently requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
endforeach()
add_compile_definitions(UNICODE _UNICODE)
add_compile_options(/W4 /WX /utf-8 /permissive-)
add_link_options(/DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA)
add_library(StellarisCursorHook SHARED src/cursor_hook.cpp)
target_link_libraries(StellarisCursorHook PRIVATE user32 gdi32 shell32 ole32 advapi32)
add_executable(StellarisCursor WIN32 src/controller.cpp)
target_link_libraries(StellarisCursor PRIVATE user32 gdi32 shell32 ole32 advapi32 comdlg32 comctl32 dwmapi gdiplus uxtheme)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_custom_target(CursorThemes ALL
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/make_themes.py" "${CMAKE_BINARY_DIR}/bin/assets/themes"
  BYPRODUCTS "${CMAKE_BINARY_DIR}/bin/assets/themes/1/normal.cur"
  COMMENT "Generating original semantic cursor themes")
add_dependencies(StellarisCursor CursorThemes)
set(EMBEDDED_DIR "${CMAKE_BINARY_DIR}/embedded")
add_custom_command(OUTPUT "${EMBEDDED_DIR}/runtime.rc" "${EMBEDDED_DIR}/runtime_manifest.h"
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/embed_runtime.py"
    --dll "$<TARGET_FILE:StellarisCursorHook>" --themes "${CMAKE_BINARY_DIR}/bin/assets/themes" --output "${EMBEDDED_DIR}"
  DEPENDS StellarisCursorHook CursorThemes "${CMAKE_SOURCE_DIR}/tools/embed_runtime.py" "${CMAKE_SOURCE_DIR}/tools/make_themes.py"
  COMMENT "Embedding verified original runtime for single-file launch")
add_custom_target(EmbeddedRuntime DEPENDS "${EMBEDDED_DIR}/runtime.rc" "${EMBEDDED_DIR}/runtime_manifest.h")
add_dependencies(StellarisCursor EmbeddedRuntime)
target_sources(StellarisCursor PRIVATE "${EMBEDDED_DIR}/runtime.rc")
target_include_directories(StellarisCursor PRIVATE "${EMBEDDED_DIR}")
target_compile_definitions(StellarisCursor PRIVATE CURSORBRIDGE_STANDALONE)
add_executable(StellarisCursorTest src/test_host.cpp)
target_link_libraries(StellarisCursorTest PRIVATE user32 gdi32)
add_executable(StellarisCursorBridgeTest src/bridge_test.cpp)
target_link_libraries(StellarisCursorBridgeTest PRIVATE user32 gdi32)
add_executable(StellarisCursorLanguageTest src/language_test.cpp)
target_link_libraries(StellarisCursorLanguageTest PRIVATE user32 gdi32 shell32)
add_executable(StellarisCursorPathsTest src/paths_test.cpp)
target_link_libraries(StellarisCursorPathsTest PRIVATE user32 gdi32 shell32 ole32 advapi32)
add_executable(StellarisCursorPreferencesTest src/preferences_test.cpp)
target_link_libraries(StellarisCursorPreferencesTest PRIVATE user32 gdi32)
add_executable(StellarisCursorPanelTest src/panel_test.cpp)
target_link_libraries(StellarisCursorPanelTest PRIVATE user32 gdi32 shell32 ole32 advapi32 comdlg32 comctl32 dwmapi gdiplus uxtheme)
add_executable(StellarisCursorConnectionHost WIN32 src/connection_host.cpp)
target_link_libraries(StellarisCursorConnectionHost PRIVATE user32 gdi32)
add_executable(StellarisCursorConnectionTest src/connection_test.cpp)
target_link_libraries(StellarisCursorConnectionTest PRIVATE user32 gdi32 shell32 ole32 advapi32 comdlg32 comctl32 dwmapi gdiplus uxtheme)
add_dependencies(StellarisCursorConnectionTest StellarisCursorConnectionHost StellarisCursorHook CursorThemes)
enable_testing()
add_test(NAME bridge COMMAND StellarisCursorBridgeTest)
add_test(NAME language COMMAND StellarisCursorLanguageTest "-userdir=E:\\Game Profiles\\中文")
add_test(NAME paths COMMAND StellarisCursorPathsTest)
add_test(NAME preferences COMMAND StellarisCursorPreferencesTest)
add_test(NAME panel COMMAND StellarisCursorPanelTest)
add_test(NAME connection COMMAND StellarisCursorConnectionTest)
set_tests_properties(connection PROPERTIES TIMEOUT 60)
