cmake_minimum_required(VERSION 3.21)
project(CursorBridge VERSION 0.6.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "CursorBridge currently requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
endforeach()
add_compile_definitions(UNICODE _UNICODE)
add_compile_options(/W4 /WX /utf-8 /permissive-)
add_link_options(/DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA)
add_library(StellarisCursorHook SHARED src/cursor_hook.cpp)
target_link_libraries(StellarisCursorHook PRIVATE user32 gdi32 shell32 ole32 advapi32)
add_executable(StellarisCursor WIN32 src/controller.cpp)
target_link_libraries(StellarisCursor PRIVATE user32 gdi32 shell32 ole32 advapi32 comdlg32)
add_executable(StellarisCursorTest src/test_host.cpp)
target_link_libraries(StellarisCursorTest PRIVATE user32 gdi32)
add_executable(StellarisCursorBridgeTest src/bridge_test.cpp)
target_link_libraries(StellarisCursorBridgeTest PRIVATE user32 gdi32)
add_executable(StellarisCursorLanguageTest src/language_test.cpp)
target_link_libraries(StellarisCursorLanguageTest PRIVATE user32 gdi32 shell32)
add_executable(StellarisCursorPathsTest src/paths_test.cpp)
target_link_libraries(StellarisCursorPathsTest PRIVATE user32 gdi32 shell32 ole32 advapi32)
enable_testing()
add_test(NAME bridge COMMAND StellarisCursorBridgeTest)
add_test(NAME language COMMAND StellarisCursorLanguageTest "-userdir=E:\\Game Profiles\\中文")
add_test(NAME paths COMMAND StellarisCursorPathsTest)
